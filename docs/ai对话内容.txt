你们需要调研 STUN/TURN 并画出流程图。STUN 的核心就是通过一个公网 Controller ，让隐藏在 NAT 后面的设备互相知道对方的公网出口 IP 和端口，然后同时向对方发包，在各自的路由器防火墙上“打穿”一个洞 。

2. C 语言测试代码 (test_api.c)
创建一个名为 test_api.c 的文件，并将以下代码粘贴进去。代码中使用了动态内存分配来接收完整的 JSON 响应，并将其打印到终端供你验证。

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <curl/curl.h>

// 内存结构体，用于保存从接口接收到的 JSON 数据
struct MemoryStruct {
  char *memory;
  size_t size;
};

// libcurl 的回调函数，负责将收到的数据块拼接起来
static size_t WriteMemoryCallback(void *contents, size_t size, size_t nmemb, void *userp) {
  size_t realsize = size * nmemb;
  struct MemoryStruct *mem = (struct MemoryStruct *)userp;

  char *ptr = realloc(mem->memory, mem->size + realsize + 1);
  if(!ptr) {
    printf("内存分配失败\n");
    return 0;
  }

  mem->memory = ptr;
  memcpy(&(mem->memory[mem->size]), contents, realsize);
  mem->size += realsize;
  mem->memory[mem->size] = 0;

  return realsize;
}

int main(void) {
  CURL *curl_handle;
  CURLcode res;
  struct MemoryStruct chunk;

  chunk.memory = malloc(1);
  chunk.size = 0;

  curl_global_init(CURL_GLOBAL_ALL);
  curl_handle = curl_easy_init();

  if(curl_handle) {
    // 目标接口: 2.2 获取网络拓扑节点列表
    // 注意：请将 127.0.0.1:8080 替换为你 Controller 实际运行的 IP 和端口
    const char *url = "http://127.0.0.1:8080/api/v1/dashboard/nodes";

    // 配置 curl 请求参数
    curl_easy_setopt(curl_handle, CURLOPT_URL, url);
    curl_easy_setopt(curl_handle, CURLOPT_WRITEFUNCTION, WriteMemoryCallback);
    curl_easy_setopt(curl_handle, CURLOPT_WRITEDATA, (void *)&chunk);
    curl_easy_setopt(curl_handle, CURLOPT_USERAGENT, "NetWeaver-Node-Test/1.0");

    printf("正在发起 GET 请求至: %s\n", url);
    
    // 执行 HTTP GET 请求
    res = curl_easy_perform(curl_handle);

    // 检查错误
    if(res != CURLE_OK) {
      fprintf(stderr, "请求失败: %s\n", curl_easy_strerror(res));
    } else {
      long response_code;
      curl_easy_getinfo(curl_handle, CURLINFO_RESPONSE_CODE, &response_code);
      printf("HTTP 状态码: %ld\n", response_code);
      printf("================ 响应负载 (JSON) ================\n");
      printf("%s\n", chunk.memory);
      printf("=================================================\n");
    }

    // 清理资源
    curl_easy_cleanup(curl_handle);
    free(chunk.memory);
  }

  curl_global_cleanup();
  return 0;
}

这里有一个极其关键的前端联调避坑点：你的 Vue 3 项目大概率运行在 localhost:5173，而这个 C 程序如果运行在 localhost:8080，浏览器的同源策略会直接把请求拦截（大名鼎鼎的 CORS 跨域报错）。因此，我在下面这个 C 程序的 HTTP 响应头里，专门为你加上了 Access-Control-Allow-Origin: *，确保你的前端能顺利拿到数据。

1. C 语言 Mock Server 代码 (mock_server.c)
创建一个名为 mock_server.c 的文件，填入以下代码：

C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

#define PORT 8080
#define BUFFER_SIZE 1024

// 严格按照接口文档 2.2 定义的 JSON 数据
const char *json_payload = 
    "{\n"
    "  \"code\": 200,\n"
    "  \"msg\": \"success\",\n"
    "  \"data\": {\n"
    "    \"nodes\": [\n"
    "      {\n"
    "        \"node_id\": \"nw-node-a1b2\",\n"
    "        \"hostname\": \"ubuntu-server-01\",\n"
    "        \"virtual_ip\": \"10.0.0.2\",\n"
    "        \"public_ip\": \"203.0.113.5\",\n"
    "        \"nat_type\": \"Full Cone\",\n"
    "        \"status\": \"online\",\n"
    "        \"connected_peers\": 3\n"
    "      },\n"
    "      {\n"
    "        \"node_id\": \"nw-node-c3d4\",\n"
    "        \"hostname\": \"macbook-pro\",\n"
    "        \"virtual_ip\": \"10.0.0.3\",\n"
    "        \"public_ip\": \"198.51.100.12\",\n"
    "        \"nat_type\": \"Symmetric\",\n"
    "        \"status\": \"online\",\n"
    "        \"connected_peers\": 3\n"
    "      }\n"
    "    ]\n"
    "  }\n"
    "}";

int main() {
    int server_fd, new_socket;
    struct sockaddr_in address;
    int opt = 1;
    int addrlen = sizeof(address);
    char buffer[BUFFER_SIZE] = {0};

    // 1. 创建 TCP Socket
    if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
        perror("Socket creation failed");
        exit(EXIT_FAILURE);
    }

    // 2. 设置 Socket 选项，允许端口复用（防止重启程序时报 Address already in use）
    if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR | SO_REUSEPORT, &opt, sizeof(opt))) {
        perror("setsockopt failed");
        exit(EXIT_FAILURE);
    }

    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY; // 监听所有网卡
    address.sin_port = htons(PORT);

    // 3. 绑定端口
    if (bind(server_fd, (struct sockaddr *)&address, sizeof(address)) < 0) {
        perror("Bind failed");
        exit(EXIT_FAILURE);
    }

    // 4. 开始监听
    if (listen(server_fd, 3) < 0) {
        perror("Listen failed");
        exit(EXIT_FAILURE);
    }

    printf("NetWeaver Mock Controller 启动成功，监听端口: %d\n", PORT);
    printf("等待前端 Vue 项目发起请求...\n");

    // 5. 死循环接收前端请求
    while(1) {
        if ((new_socket = accept(server_fd, (struct sockaddr *)&address, (socklen_t*)&addrlen)) < 0) {
            perror("Accept failed");
            continue;
        }

        // 读取前端发来的 HTTP 请求报文（虽然是 Mock，但我们要把缓冲区的请求读掉）
        read(new_socket, buffer, BUFFER_SIZE - 1);
        printf("\n--- 收到前端请求 ---\n%s\n", buffer);

        // 构造 HTTP 响应报文 (Header + Body)
        // 关键点：Access-Control-Allow-Origin: * 解决 Vue 前端跨域问题
        char http_response[4096];
        sprintf(http_response, 
                "HTTP/1.1 200 OK\r\n"
                "Content-Type: application/json; charset=utf-8\r\n"
                "Access-Control-Allow-Origin: *\r\n"
                "Content-Length: %zu\r\n"
                "\r\n"
                "%s", 
                strlen(json_payload), json_payload);

        // 发送给前端
        write(new_socket, http_response, strlen(http_response));
        printf("已成功返回 Mock 节点数据。\n");

        // 断开本次连接 (HTTP/1.1 短链接模式)
        close(new_socket);
        memset(buffer, 0, BUFFER_SIZE);
    }

    return 0;
}

const char *json_404 = 
    "{\n"
    "  \"code\": 404,\n"
    "  \"msg\": \"API Route Not Found\",\n"
    "  \"data\": {}\n"
    "}";

// ==================== 2. HTTP 响应封装函数 ====================

// 专门用来发送 HTTP 响应，自动计算 Content-Length 并注入 CORS 头
void send_json_response(int socket_fd, int http_status, const char* status_text, const char* json_payload) {
    char http_response[4096];
    sprintf(http_response, 
            "HTTP/1.1 %d %s\r\n"
            "Content-Type: application/json; charset=utf-8\r\n"
            "Access-Control-Allow-Origin: *\r\n"
            "Content-Length: %zu\r\n"
            "\r\n"
            "%s", 
            http_status, status_text, strlen(json_payload), json_payload);

    write(socket_fd, http_response, strlen(http_response));
}

// ==================== 3. 主程序与路由逻辑 ====================

int main() {
    int server_fd, new_socket;
    struct sockaddr_in address;
    int opt = 1;
    int addrlen = sizeof(address);
    char buffer[BUFFER_SIZE] = {0};

    // 创建、设置并绑定 Socket (基础网络编程三步曲)
    if ((server_fd = socket(AF_INET, SOCK_STREAM, 0)) == 0) {
        perror("Socket creation failed"); exit(EXIT_FAILURE);
    }
    if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR | SO_REUSEPORT, &opt, sizeof(opt))) {
        perror("setsockopt failed"); exit(EXIT_FAILURE);
    }
    address.sin_family = AF_INET;
    address.sin_addr.s_addr = INADDR_ANY;
    address.sin_port = htons(PORT);
    if (bind(server_fd, (struct sockaddr *)&address, sizeof(address)) < 0) {
        perror("Bind failed"); exit(EXIT_FAILURE);
    }
    if (listen(server_fd, 5) < 0) {
        perror("Listen failed"); exit(EXIT_FAILURE);
    }

    printf("NetWeaver Mock Controller (Router 升级版) 启动成功!\n");
    printf("监听端口: %d | 等待前端请求...\n", PORT);
    printf("支持的路由:\n");
    printf("  - GET /api/v1/dashboard/stats\n");
    printf("  - GET /api/v1/dashboard/nodes\n\n");

    while(1) {
        if ((new_socket = accept(server_fd, (struct sockaddr *)&address, (socklen_t*)&addrlen)) < 0) {
            perror("Accept failed"); continue;
        }

        memset(buffer, 0, BUFFER_SIZE);
        read(new_socket, buffer, BUFFER_SIZE - 1);

        // 核心逻辑：解析 HTTP 请求的首行
        char method[16] = {0};
        char path[256] = {0};
        // 使用 sscanf 提取按空格分隔的前两个字符串 (例如 "GET" 和 "/api/...")
        sscanf(buffer, "%15s %255s", method, path);

        if (strlen(method) > 0 && strlen(path) > 0) {
            printf("[Mock 收到请求] Method: %s, Path: %s\n", method, path);

            // ==================== 路由分发 (Router) ====================
            if (strcmp(method, "GET") == 0 && strcmp(path, "/api/v1/dashboard/stats") == 0) {
                // 命中 2.1 统计大屏接口
                send_json_response(new_socket, 200, "OK", json_stats);
                printf("  -> 已返回 Stats 统计数据\n");

            } else if (strcmp(method, "GET") == 0 && strcmp(path, "/api/v1/dashboard/nodes") == 0) {
                // 命中 2.2 节点拓扑接口
                send_json_response(new_socket, 200, "OK", json_nodes);
                printf("  -> 已返回 Nodes 节点列表\n");

            } else if (strcmp(method, "OPTIONS") == 0) {
                // 处理浏览器发出的 CORS 预检请求 (Preflight)
                char preflight_response[512];
                sprintf(preflight_response, 
                        "HTTP/1.1 204 No Content\r\n"
                        "Access-Control-Allow-Origin: *\r\n"
                        "Access-Control-Allow-Methods: GET, POST, OPTIONS\r\n"
                        "Access-Control-Allow-Headers: Content-Type, Authorization\r\n"
                        "\r\n");
                write(new_socket, preflight_response, strlen(preflight_response));
                printf("  -> 已放行 OPTIONS 预检请求\n");

            } else {
                // 路由未命中，返回 404
                send_json_response(new_socket, 404, "Not Found", json_404);
                printf("  -> 路由不存在，返回 404\n");
            }
        }

        close(new_socket); // 短链接，响应完毕即关闭
    }

    return 0;
}
代码设计的核心亮点
灵活的路由匹配： 使用 strcmp(path, "/api/v1/dashboard/stats") 替代了写死数据的做法。未来如果需要添加 3.3 拉取对等节点路由表 GET /api/v1/nodes/{node_id}/peers 等测试接口给后端节点用，只需在 if-else 里再加一个分支即可。  

优雅的 send_json_response 封装： 将拼接 HTTP Header 的脏活累活抽离成了公共函数，避免了大量的代码复制。这里使用 strlen() 动态计算 Content-Length，非常规范。

处理了 OPTIONS 预检请求： 当 Vue 3 使用 Axios 发送带请求头（如 Token）的 AJAX 请求时，浏览器会先发送一个 OPTIONS 方法的请求去探路。这段代码专门增加了对 OPTIONS 的放行，这是彻底解决复杂跨域问题的终极杀招。

在 WSL 里直接 gcc mock_router.c -o mock_router 编译运行即可。现在前端团队可以在代码里同时发起请求这两个接口，并在同一个大屏页面上渲染数字仪表盘和网络拓扑图了。

你觉得我们节点之间有必要指定关系吗

这是一个非常有深度的架构问题。对于一个异地组网（SDN/P2P）平台来说，绝对有必要指定关系，但我们需要明确“关系”的具体维度。

在你们目前的开发阶段（MVP阶段）和未来的企业级演进中，“节点关系”其实分为连通性关系（必须有）和逻辑权限关系（未来有）。

我建议你们分阶段来看待这个问题：

1. 当前阶段必须有的关系：物理/连通性拓扑 (Network State)
NetWeaver 的核心卖点就是“P2P 架构”。如果节点之间没有明确的连通关系数据，你们前端的 Echarts 拓扑图就只能画出一堆散落的孤岛，或者全部连向 Controller 的“星型”结构（这就退化成传统 VPN 了）。

你们必须在 Controller 中维护并下发以下关系：

直连关系 (P2P Tunnel)： Node A 和 Node B 通过 UDP 打洞成功了，流量不经过 Controller。

中转关系 (Relay Fallback)： Node A 和 Node C 打洞失败（比如都是对称型 NAT），只能通过 Controller 或者某个中继节点转发流量。

落地建议：
在你们的 GET /api/v1/dashboard/nodes 接口或者专门的拓扑接口中，应该引入 links（边）的概念。前端 Echarts 画图需要 nodes（点）和 links（边）。

JSON
"data": {
  "nodes": [ ... ],
  "links": [
    {
      "source": "nw-node-a1b2",
      "target": "nw-node-c3d4",
      "link_type": "p2p",     // 关系类型：直连
      "latency_ms": 45        // 附加属性：延迟
    },
    {
      "source": "nw-node-a1b2",
      "target": "controller-01",
      "link_type": "relay",   // 关系类型：中转
      "latency_ms": 120
    }
  ]
}
2. 下一阶段可以引入的关系：逻辑隔离与权限 (Access Control)
如果是打比赛，为了突出“企业级”和“安全性”，你们可以在 Controller 层面引入逻辑关系。默认情况下，所有接入的节点都在同一个 10.0.0.0/24 网段，互相能 Ping 通。但在真实场景中，这很危险（零信任网络架构不希望这样）。

租户/网络组 (Network Groups)： Node A 和 Node B 属于“开发组”，Node C 属于“财务组”。Controller 在下发路由表时，只把同组的节点信息发给对方。

出站/入站规则 (ACLs)： 允许 Node A 访问 Node B 的 80 端口，但不允许 Node B 主动访问 Node A。

3. 高级架构的关系：节点角色 (Node Roles)
你们目前的架构是：所有节点地位平等，打不通的流量全靠公网 Controller 中转。这会导致公网服务器带宽压力巨大。
更高级的做法是赋予节点特定的关系角色：

边缘节点 (Edge Node)： 普通的 PC 或工控机。

超级节点 (Super/Derp Node)： 如果某个普通节点碰巧拥有公网 IP（比如一台部署在阿里云的普通服务器也装了你们的客户端），Controller 可以动态把它指定为“中继节点”，让其他打洞失败的节点走它转发。这就是 Tailscale 的核心玩法。

总结给你们目前的行动指引：
不要一开始就把权限隔离和超级节点搞得太复杂。这周末的联调，只需让 Controller 知道谁和谁打洞成功了，把这个 "P2P 还是 Relay" 的关系传给前端即可。

在你们目前的 Echarts 拓扑图设计中，你们打算如何通过视觉元素（比如颜色、线条样式、动态效果）来直观地向评委区分“P2P 高速直连”和“服务器低速中转”这两种不同的网络关系呢？

前后端在校园网联调的时候出现请求超时怎么办

在校园网环境下进行前后端联调，“请求超时”（Timeout）简直是家常便饭。校园网的网络拓扑通常极其复杂，并且伴随着严格的安全策略。

遇到这个问题，不要盲目改代码，我们需要从应用层到网络层进行**“四步排查法”**。请你和前端同学按照以下顺序逐一核对：

1. 检查后端监听地址（最容易犯的错）
如果你的后端（Controller）代码是用 Gin 写的，检查你启动服务的代码。

错误写法： r.Run("127.0.0.1:8080") 或 r.Run("localhost:8080")。这表示后端只接受来自这台电脑内部的请求。

正确写法： r.Run(":8080") 或 r.Run("0.0.0.0:8080")。这表示后端监听所有网卡，允许局域网内其他机器访问。

2. 检查前端请求的 IP 地址
前端 Vue 项目在发起 Axios 请求时，URL 绝对不能写 localhost 或 127.0.0.1。

排查动作： 在运行后端的那台电脑上，打开终端输入 ipconfig (Windows) 或 ip a (Linux)，找到校园网分配给这台机器的真实局域网 IP（比如 10.x.x.x 这种）。

前端代码里的 baseURL 必须换成这个真实的 IP，例如：[http://10.64.23.11:8080/api/v1/dashboard/nodes](http://10.64.23.11:8080/api/v1/dashboard/nodes)。

3. 避开 WSL2 的“网络黑洞”陷阱
如果你的 Controller 服务是跑在 WSL2 里面的，这就非常关键了。WSL2 本质上是在 Windows 里套了一层 NAT 虚拟网络（它的 IP 通常是 172.x.x.x）。
你的前端同学如果在另一台电脑上，他是绝对无法直接访问到你 WSL2 里的端口的，请求一定会超时。

解决方案 A（原生映射）： 在 Windows 宿主机上以管理员身份打开 PowerShell，使用 netsh interface portproxy 命令，把 Windows 的 8080 端口流量转发给 WSL2 的 8080 端口。

解决方案 B（最省事）： 联调期间，直接把编译好的 Go Windows 二进制文件 (.exe) 放到 Windows 宿主机上运行，暂时跳过 WSL2。

4. 校园网 AP 隔离与防火墙拦截（终极杀手）
如果以上都对了还是超时，那基本就是网络环境的问题了。

Windows 防火墙： 运行后端的电脑必须在“Windows Defender 防火墙”的“高级设置”中，新建一个“入站规则”，放行 TCP 的 8080 端口。

校园网 AP 隔离： 为了防范病毒传播，很多学校的 Wi-Fi 开启了“AP 隔离（客户端隔离）”。这就导致即使你们连着同一个校园 Wi-Fi，设备之间也是不能互相 Ping 通的。

终极突围方案：
如果确认是校园网搞了 AP 隔离，你们有两个选择：

物理突围： 你们两个人连同一个手机开的 5G 热点，手机热点通常没有局域网隔离，立马就能通。

公网突围： 如果手机流量不够，可以直接使用 SakuraFrp 这类内网穿透工具，把你的本地 Controller 端口映射到一个公网域名上，让前端直接请求公网地址进行联调。

你们现在的 Controller 是跑在 WSL2 里，还是直接在 Windows 宿主机上运行的？

以下是为后续任务补充的**【用户视角】**解读，帮你们团队在埋头写代码时，始终明白自己正在创造什么价值：

Phase 3: 隧道封装与真实业务接管 (第 5-6 周)
Week 5: L3 网络报文的“偷天换日” (TUN + UDP 封装)
研发动作： 将截获的虚拟网卡 IP 报文作为 Payload，塞入 P2P 的 UDP 隧道发送并解包。

【用户视角：跨越物理隔离的“虫洞”】

用途理解： 在这周之前，前端面板上画的拓扑线只是“概念上的连通”。而这周干完后，奇迹发生了：用户在学校宿舍的笔记本上，可以直接通过浏览器访问放在家里（没有公网 IP、连在路由器后）的 NAS 影视库。

体验变化： 用户感知不到底层发生了多么复杂的拆包组装，他们只知道：“只要我接入了 NetWeaver，异地的设备就像插在了同一台交换机上一样，IP 互通了。”

Week 6: 弱网对抗与传输层重构 (引入 KCP)
研发动作： 剥离原生 UDP，接入 KCP 协议并调优 RTO/拥塞控制策略；引入 tc 丢包模拟。

【用户视角：高铁与星巴克里的“定海神针”】

用途理解： 真实的网络不是实验室里的千兆局域网。当用户在高铁上或者连接着拥挤的咖啡厅 Wi-Fi 时，网络会有极高的延迟和随机丢包。传统的 VPN 在这种情况下会疯狂断线重连，远程桌面会卡成幻灯片，SSH 敲一个字母要等三秒。

体验变化： KCP 优化完成后，用户会发现：“即使我的物理网络丢包率高达 20%，通过 NetWeaver 连接的远程桌面依然丝滑，SSH 敲击零延迟。” 这是产品体验的绝对护城河。

Phase 4: 安全加固、部署与交付 (第 7-8 周)
Week 7: 端到端加密与一键部署 (AES-GCM & Docker/Shell)
研发动作： 在 KCP 传输前实施数据流对称加密；编写自动化部署脚本和 Docker 镜像。

【用户视角：绝对的隐私与“傻瓜式”体验】

用途理解（加密）： 企业用户最怕“数据出境”。由于你们采用了端到端加密，这意味着：即便是部署在阿里云上的信令 Controller（甚至是你们开发者本人），也绝对无法偷窥、抓包或者篡改用户传输的机密文件。 Controller 只是个“红娘”，绝不触碰用户的“情书”。

用途理解（部署）： 传统开源组网工具配置极度反人类，需要手写路由表。一键部署脚本完成后，用户只需在终端复制粘贴一行代码，敲下回车，剩下的网卡创建、权限提升、连接服务器全自动完成。

体验变化： “不用学 Linux 网络基础，小白也能在一分钟内建好企业级私有网。”

Week 8: 决战包装与业务级演示
研发动作： 停止代码开发，全力准备演示用例（如 RDP、NAS 挂载）和答辩 PPT。

【用户视角：原来它可以做这么多事！】

用途理解： 评委和最终用户一样，他们不想看满屏翻滚的 Linux 终端日志。当你们在演示视频中，不用任何向日葵/TeamViewer，直接用 Windows 原生的 RDP 连接上异地的电脑打游戏，或者直接在本地的 VS Code 里挂载并修改远端服务器的代码时，价值就彻底具象化了。

体验变化： 这不再是一个“网络实验”，而是一个能真正改变工作方式的“生产力工具底座”。

给团队的沟通建议：
在周五和周日的 Check-in 会议上，作为 PM，在询问“代码写得怎么样了”之前，先问一句：“兄弟们，我们这周的进度，能让用户在星巴克里更爽地连上实验室的电脑了吗？” 这能极大地凝聚团队的方向感。

现在在已知在这几周要交付的产品的效果的基础上，告诉我每个人那几周分别相应的时候应该完成了哪些内容（包括之前做的和那周做的）（要具体到功能逻辑甚至是文件）

这份基于时间的“交付物快照清单”，将精确到具体的功能表现和核心代码文件。这种视角的转换能极大消除开发中的不确定性。你们可以用这份清单作为每次双周 Check-in 的**“验收军令状”**。

以下是按关键时间节点（每两周为一个里程碑），团队三人累积应完成的详细状态：

里程碑 1：截止第二周末（基础设施打通期）
此时的产品状态： 一个能跑起来的“空壳”。公网服务器能接收请求，边缘电脑能创建虚拟网卡，网页端能看到管理后台的静态布局。

1. 后端研发
功能状态： Controller 已部署并暴露 HTTP 接口；Node 客户端已能在本地生成虚拟网卡，终端可打印抓取到的包。

文件级交付：

cmd/controller/main.go：信令服务器的启动入口。

internal/controller/api/router.go：包含用 Gin 写的 /api/v1/nodes/register 和 /api/v1/dashboard/nodes。

internal/node/tun/device.go：封装了 water 库的代码，成功在宿主机创建出 tun0。

2. 前端研发
功能状态： 抛弃了 C 语言 Mock Server，已经能用 Axios 请求并解析 Controller 的真实数据。页面有导航栏和数据表格。

文件级交付：

src/api/request.ts：Axios 拦截器配置，设置了统一的 BaseURL。

src/views/Dashboard.vue：主面板页面，使用 Element Plus 的 Layout 布局。

src/components/NodeTable.vue：节点列表组件，能渲染后端的 JSON 数组。

3. PM/文档
功能状态： 统一了开发语言，搭好了文档架子。

文件级交付：

NetWeaver_API_v1.md：详细的 RESTful 接口契约，前后端完全遵循此文档联调。

thesis_main.tex：LaTeX 论文主文件，包含封面、摘要、目录，以及由 AI 辅助扩写的“1. 绪论”部分。

里程碑 2：截止第四周末（P2P 穿透与可视化期）
此时的产品状态： 骨架上长出了肌肉。两台边缘设备通过公网互相找到了对方的 IP 并在底层尝试通信；网页端出现了一张非常酷炫的网络拓扑连线图。

1. 后端研发
功能状态： Controller 实现内存级并发安全的节点管理；Node 实现了基础 STUN 探测和互相发送 UDP 探测包。

文件级交付：

internal/controller/state/registry.go：包含一个 map[string]*Node 和读写锁 sync.RWMutex，确保高并发注册不崩溃。

internal/node/p2p/stun.go：利用 UDP 向公网发包并解析自己 NAT 外网 IP 的逻辑。

internal/node/p2p/punch.go：核心打洞逻辑，互相向对方的公网 <IP:Port> 轰炸 UDP 字符串。

2. 前端研发
功能状态： 之前的表格数据被图形化了。如果 Controller 告诉前端有三个节点，页面上就有三个圆圈，并且画出了连线。

文件级交付：

src/components/TopologyGraph.vue：极其核心的 Echarts 或 G6 组件。实现了将 API 里的节点转为 nodes，将打洞成功的关系转为 links 进行渲染。

3. PM/文档
功能状态： 完成底层逻辑的理论沉淀。

文件级交付：

diagrams/nat_traversal.drawio：一张精准的 STUN 穿透时序图。

Postman Workspace：里面录入了 Controller 的所有 API，可以一键 Run 测试服务端是否正常。

里程碑 3：截止第六周末（业务接管与弱网对抗期 —— 决定成败的关键）
此时的产品状态： 真正的跨地域组网完成！在宿舍的电脑上打开终端，ping 10.0.0.3（实验室电脑的虚拟IP），不仅通了，而且即便用工具限制网速制造丢包，延迟依然稳定。

1. 后端研发
功能状态： 虚拟网卡与 UDP/KCP 隧道正式合体。内存零拷贝优化上线。

文件级交付：

internal/node/forwarder.go：极其核心的数据流转中枢。它包含一个死循环：从 tun 读取字节 -> 扔进 kcp 隧道发出；从 kcp 收到字节 -> 写入 tun。

pkg/pool/buffer.go：引入 sync.Pool 的内存池配置，避免高频发包导致 GC 卡顿。

2. 前端研发
功能状态： 页面不再是静态的，不仅能看结构，还能看性能监控。

文件级交付：

src/components/TrafficLineChart.vue：基于 Echarts 的实时折线图，展示当前网络的上下行吞吐量（KB/s）和延迟（RTT）。

src/components/NodeDrawer.vue：点击拓扑图上的节点，右侧滑出抽屉，展示该节点的详细报表。

3. PM/文档
功能状态： 完成压力测试与数据收集。

文件级交付：

test_scripts/simulate_weak_net.sh：使用 Linux tc 命令配置 20% 丢包的脚本。

perf_report.md：收集开启 KCP 前后、有无丢包情况下的 Ping 延迟和 Iperf3 带宽测试数据，整理成表格和折线图。

里程碑 4：截止第八周末（交付与包装冲刺期）
此时的产品状态： 工业级成品。流量被加密，产品可一键安装，拥有一份逻辑严密、排版精美的参赛论文，以及一段极具震撼力的演示视频。

1. 后端研发
功能状态： 补齐企业级安全属性，消除硬编码。

文件级交付：

internal/node/crypto/cipher.go：AES-GCM 端到端加密逻辑。

configs/config.yaml：将原本写死在代码里的 Controller IP、端口、加密密钥全部提取为配置文件。

Makefile：提供 make build-linux 和 make build-win 跨平台编译脚本。

2. 前端研发
功能状态： 代码封板，前端具备生产环境的部署能力。

文件级交付：

Dockerfile：使用 Node.js 镜像 build，随后用 Nginx 镜像暴露 80 端口。

nginx.conf：配置静态资源缓存和 Vue Router 必须的跨域代理。

3. PM/文档
功能状态： 所有的技术结晶转化为最终的答辩物料。

文件级交付：

install_node.sh：一键安装脚本（自动拉取二进制文件并配置 systemd 守护进程）。

《NetWeaver 网络挑战赛路演.pptx》：突出“SDN解耦”、“KCP弱网优化”和“自研零黑盒”三大卖点。

Demo.mp4：实录视频。展示在弱网环境下，通过 NetWeaver 建立隧道，极其流畅地使用 RDP 远程桌面或者挂载远程代码仓库的操作。
